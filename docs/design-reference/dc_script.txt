  const m10 = n % 10, m100 = n % 100;
  if (m10 === 1 && m100 !== 11) return one;
  if (m10 >= 2 && m10 <= 4 && (m100 < 12 || m100 > 14)) return few;
  return many;
};

  state = {
    screen: null,
    cat: 'all',
    sort: 'pop',
    onlyFree: false,
    productId: 'a7siii',
    cart: [{ id: 'a7siii', qty: 1 }, { id: 'hero13', qty: 2 }],
    dateFrom: 13,
    dateTo: 15,
    timeFrom: '11:00',
    timeTo: '18:00',

    pickerTarget: 'from',
    promoIndex: 0,
    form: { name: '', phone: '', contact: '', comment: '' },
    agreeData: false,
    agreePolicy: false,
    submitted: false,
    adminTab: 'orders'
  };

  componentDidMount() {
    this.timer = window.setInterval(() => {
      if ((this.props.promoAuto ?? true) && !this.state.pickerOpen) {
        this.setState((p) => ({ promoIndex: (p.promoIndex + 1) % PROMOS.length }));
      }
    }, 9000);
  }
  componentWillUnmount() { window.clearInterval(this.timer); }

  setField = (k) => (e) => {
    const v = e.target.value;
    this.setState((p) => ({ form: { ...p.form, [k]: v }, submitted: false }));
  };

  go = (screen) => { this.setState({ screen, pickerOpen: false }); window.scrollTo(0, 0); };
  openItem = (id) => { this.setState({ screen: 'product', productId: id, pickerOpen: false }); window.scrollTo(0, 0); };
  add = (id, qty) => {
    this.setState((s) => {
      const cart = s.cart.slice();
      const hit = cart.find((l) => l.id === id);
      if (hit) hit.qty += (qty || 1); else cart.push({ id, qty: qty || 1 });
      return { cart, submitted: false };
    });
  };

  fmtDate = (d) => d + ' августа';

    const fee = this.props.offHoursFee ?? 2500;
    const showKits = this.props.showKits ?? true;
    const s = { ...this.state, screen: this.state.screen || (this.props.defaultScreen ?? 'home') };
    const item = ALL.find((i) => i.id === s.productId) || ALL[0];
    const days = Math.max(1, s.dateTo - s.dateFrom);
    const isOff = (t) => {
      const h = parseInt(t.slice(0, 2), 10) + parseInt(t.slice(3), 10) / 60;
      return h < 10 || h > 21;
    };
    const offHours = isOff(s.timeFrom) || isOff(s.timeTo);
    const slotOf = (id) => (KITS.some((k) => k.id === id) ? 'pb-kit-' + id : 'pb-item-' + id);
    const cartLines = s.cart.map((l) => {
      const it = ALL.find((i) => i.id === l.id);
      return { l, it, sum: it.price * l.qty * days };
    });
    const cartTotal = cartLines.reduce((a, x) => a + x.sum, 0) + (offHours ? fee : 0);
    const POOL = showKits ? ALL : ITEMS;

    let list = POOL.filter((i) => (s.cat === 'all' || i.cat === s.cat));
    if (s.onlyFree) list = list.filter((i) => i.free);
    if (s.sort === 'asc') list = list.slice().sort((a, b) => a.price - b.price);
    if (s.sort === 'desc') list = list.slice().sort((a, b) => b.price - a.price);

    const rowSel = (on) => on
      ? 'background:#0A0A0A;color:#FFFFFF'
      : 'background:#F9F8F7;color:#0A0A0A';
    const availStyle = (free) => free ? 'color:#0A0A0A' : 'color:#D62410';
    const availChip = (free) => free
      ? 'background:rgba(255,255,255,0.88);color:#0A0A0A'
      : 'background:#D62410;color:#fff';

      for (let i = 0; i < 5; i++) cells.push({ label: '', style: 'background:transparent', pick: () => {} });

        const selected = d >= s.dateFrom && d <= s.dateTo;
        const edge = d === s.dateFrom || d === s.dateTo;
        const isBusy = busy.indexOf(d) >= 0;
        let st = 'background:#F4F3F1;color:#0A0A0A;cursor:pointer';
        if (isBusy) st = 'background:rgba(214,36,16,0.12);color:#D62410;cursor:not-allowed;text-decoration:line-through';

      }
      return cells;
